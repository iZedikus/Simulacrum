│   │           │
│   │           │     -- merchant_category_code (id SMALLINT PK, code INTEGER UNIQUE)

│   │           │

│   │           │     -- financial_institution_identification_code (id SMALLINT PK, code VARCHAR UNIQUE)
│   │           │     -- Данные: BIC, LEI
│   │           │

│   │           ├── V3__create_account_table.sql
│   │           │     CREATE TABLE simulacrum.account (
│   │           │       account_id            VARCHAR(255) PRIMARY KEY,
│   │           │       status_id             SMALLINT NOT NULL REFERENCES account_status(id),
│   │           │       account_type_id       SMALLINT NOT NULL REFERENCES account_type(id),
│   │           │       status_update_datetime TIMESTAMP NOT NULL,
│   │           │       currency              CHAR(3) NOT NULL,
│   │           │       account_description   VARCHAR(255)
│   │           │     );
│   │           │
│   │           ├── V4__create_consent_table.sql
│   │           │     CREATE TABLE simulacrum.consent (
│   │           │       consent_id           UUID PRIMARY KEY DEFAULT gen_random_uuid(),
│   │           │       account_id           VARCHAR(255) NOT NULL REFERENCES simulacrum.account,
│   │           │       status               VARCHAR(30)  NOT NULL,
│   │           │       total_debit_limit    NUMERIC(15,2),
│   │           │       max_single_debit     NUMERIC(15,2),
│   │           │       currency             CHAR(3) NOT NULL,
│   │           │       purpose_code         VARCHAR(35),
│   │           │       creditor_system_id   VARCHAR(255),
│   │           │       granted_at           TIMESTAMP NOT NULL,
│   │           │       expires_at           TIMESTAMP,
│   │           │       revoked_at           TIMESTAMP
│   │           │     );
│   │           │
│   │           ├── V5__create_transaction_tables.sql
│   │           │     CREATE TABLE simulacrum.transaction_history (
│   │           │       transaction_id          VARCHAR(255) PRIMARY KEY,
│   │           │       account_id              VARCHAR(255) NOT NULL REFERENCES simulacrum.account,
│   │           │       status_id               SMALLINT NOT NULL REFERENCES transaction_status(id),
│   │           │       bank_transaction_code_id SMALLINT NOT NULL REFERENCES bank_transaction_code(id),
│   │           │       booking_date_time        TIMESTAMP NOT NULL,
│   │           │       value_date_time          TIMESTAMP NOT NULL,
│   │           │       charge_amount            NUMERIC(15,2),
│   │           │       charge_currency_id       SMALLINT REFERENCES currency(currency_id),
│   │           │       -- Debtor (встроенный)
│   │           │       debtor_name              VARCHAR(140),
│   │           │       debtor_street            VARCHAR(70),
│   │           │       debtor_building          VARCHAR(16),
│   │           │       debtor_post_code         VARCHAR(16),
│   │           │       debtor_town              VARCHAR(35),
│   │           │       debtor_country           CHAR(2),
│   │           │       -- Creditor (встроенный)
│   │           │       creditor_name            VARCHAR(140),
│   │           │       creditor_street          VARCHAR(70),
│   │           │       creditor_building        VARCHAR(16),
│   │           │       creditor_post_code       VARCHAR(16),
│   │           │       creditor_town            VARCHAR(35),
│   │           │       creditor_country         CHAR(2),
│   │           │       -- Merchant info
│   │           │       merchant_name            VARCHAR(70),
│   │           │       merchant_id              VARCHAR(35),
│   │           │       merchant_category_code_id SMALLINT REFERENCES merchant_category_code(id),
│   │           │       -- Debtor account
│   │           │       debtor_account_name      VARCHAR(70),
│   │           │       debtor_account_scheme_id  SMALLINT REFERENCES account_identification_code(id),
│   │           │       debtor_account_ident     VARCHAR(35),
│   │           │       -- Creditor account
│   │           │       creditor_account_name    VARCHAR(70),
│   │           │       creditor_account_scheme_id SMALLINT REFERENCES account_identification_code(id),
│   │           │       creditor_account_ident   VARCHAR(35),
│   │           │       -- Remittance
│   │           │       remittance_unstructured  VARCHAR(140),
│   │           │       -- Card transaction FK (nullable — не все транзакции карточные)
│   │           │       card_transaction_id      VARCHAR(255) REFERENCES simulacrum.card_transaction
│   │           │     );
│   │           │

│   │           │       additional_card_data VARCHAR(70),
